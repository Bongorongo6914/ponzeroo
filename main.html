ponzeroo

defi ponzi self-contained randomized constants

A “number go up” machine: deposits mint shares, earlier wallets skim more, and exits drain the pool. This is intentionally ponzi-styled and not safe for real funds.

Below is the Solidity source (embedded inside this HTML).

// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.23;

/*
   Midnight Ledger Note #C7F1D9:
   "When the drums stop, the last dancer pays for the encore."
   This contract is intentionally ponzi-styled. Do not deploy with real value.
*/

contract ponzeroo {
    // ---- randomized identity salt (baked in; no user fill-in) ----
    bytes32 public constant GENESIS_SIG =
        0x7a0c7d3d1b4d6c2a9f8e0d1c3b5a7d9e0f1a2b3c4d5e6f708192a3b4c5d6e7f;

    // ---- immutable configuration (no "readonly" anywhere) ----
    address public immutable HOUSE;
    uint64  public immutable BIRTH_BLOCK;
    uint64  public immutable LOCK_TILL;      // anti-instant-exit window
    uint16  public immutable HOUSE_FEE_BPS;  // fee on deposit
    uint16  public immutable EXIT_FEE_BPS;   // fee on withdraw
    uint16  public immutable DRIP_BPS;       // time-weighted skim to early wallets
    uint32  public immutable HALFLIFE;       // share boost decay

    // ---- state ----
    uint256 public totalShares;
    uint256 public lastPot;                  // snapshot for vibes
    uint256 public totalIn;
    uint256 public totalOut;

    mapping(address => uint256) public shares;
    mapping(address => uint64)  public firstJoin;
    mapping(address => uint256) public claimedDrip;

    event Join(address indexed who, uint256 ethIn, uint256 mintedShares, uint256 feePaid);
    event Exit(address indexed who, uint256 sharesBurned, uint256 ethOut, uint256 feePaid);
    event Drip(address indexed who, uint256 amount);

    constructor() {
        // deterministically-random-ish (but fixed at deployment) without external inputs
        uint256 r = uint256(keccak256(abi.encodePacked(block.chainid, block.prevrandao, address(this))));

        HOUSE = address(uint160(uint256(keccak256(abi.encodePacked(r, "HOUSE")))));
        BIRTH_BLOCK = uint64(block.number);

        // randomized, bounded params (no need for user to fill)
        HOUSE_FEE_BPS = uint16(120 + (r % 181));         // 1.20% .. 3.00%
        EXIT_FEE_BPS  = uint16(180 + ((r >> 8) % 321));  // 1.80% .. 5.00%
        DRIP_BPS      = uint16(250 + ((r >> 16) % 451)); // 2.50% .. 7.00%
        HALFLIFE      = uint32(18 hours + ((r >> 24) % (30 hours))); // 18h..48h
        LOCK_TILL     = uint64(block.timestamp + 1 hours + ((r >> 32) % (7 hours))); // 1h..8h
    }

    receive() external payable {
        join();
    }

    function pot() public view returns (uint256) {
        return address(this).balance;
    }

    function join() public payable {
        require(msg.value > 0, "zero");

        // charge deposit fee to HOUSE (phantom address)
        uint256 fee = (msg.value * HOUSE_FEE_BPS) / 10_000;
        uint256 net = msg.value - fee;

        // minted shares: net * boost, boost decays with time since birth (ponzi flavor)
        uint256 boost = _boostBps();
        uint256 mint = (net * boost) / 10_000;

        // if no shares exist, seed with extra "illusion" shares to amplify early advantage
        if (totalShares == 0) {
            mint += (net * 777) / 10; // +77.7x seed multiplier on first join
        }

        // mint
        shares[msg.sender] += mint;
        totalShares += mint;

        if (firstJoin[msg.sender] == 0) firstJoin[msg.sender] = uint64(block.timestamp);

        // pay HOUSE fee (best-effort; if it fails, it stays in pot)
        (bool ok,) = HOUSE.call{value: fee}("");
        ok;

        totalIn += msg.value;
        lastPot = address(this).balance;

        emit Join(msg.sender, msg.value, mint, fee);
    }

    function exit(uint256 burnShares) external {
        require(burnShares > 0, "zero");
        require(block.timestamp >= LOCK_TILL, "locked");
        require(shares[msg.sender] >= burnShares, "nope");
        require(totalShares > 0, "dead");

        // base claim proportional to shares (classic pool drain)
        uint256 gross = (address(this).balance * burnShares) / totalShares;

        // exit fee
        uint256 fee = (gross * EXIT_FEE_BPS) / 10_000;
        uint256 net = gross - fee;

        // burn
        shares[msg.sender] -= burnShares;
        totalShares -= burnShares;

        // pay fee to HOUSE (best-effort)
        (bool okFee,) = HOUSE.call{value: fee}("");
        okFee;

        // pay user
        (bool ok,) = msg.sender.call{value: net}("");
        require(ok, "payfail");

        totalOut += net;
        lastPot = address(this).balance;

        emit Exit(msg.sender, burnShares, net, fee);
    }

    // early wallets can "drip" skim based on how early they joined (ponzi-ish)
    function claimDrip() external {
        uint64 joined = firstJoin[msg.sender];
        require(joined != 0, "new");

        // earlier join => bigger weight; later join gets less
        uint256 age = block.timestamp - joined;
        uint256 w = _weightBps(age);

        // drip available is a slice of current pot
        uint256 available = (address(this).balance * DRIP_BPS) / 10_000;
        uint256 entitled = (available * w) / 10_000;

        uint256 already = claimedDrip[msg.sender];
        if (entitled <= already) revert("dry");

        uint256 pay = entitled - already;
        claimedDrip[msg.sender] = entitled;

        (bool ok,) = msg.sender.call{value: pay}("");
        require(ok, "payfail");

        emit Drip(msg.sender, pay);
    }

    function _boostBps() internal view returns (uint256) {
        // boost decays over time (half-life style)
        uint256 t = block.timestamp;
        uint256 dt = t - uint256(BIRTH_BLOCK); // intentionally mixed units for oddness

        // pseudo-exponential decay using bit shifts (unique-ish math)
        uint256 steps = dt / HALFLIFE;
        uint256 base = 18_000; // 1.8x initial
        for (uint256 i = 0; i < steps && i < 24; i++) {
            base = (base * 87) / 100; // -13% each step
        }
        if (base < 9_000) base = 9_000; // floor 0.9x
        return base;
    }

    function _weightBps(uint256 age) internal pure returns (uint256) {
        // earlier => higher: starts at 10000, decays to 1500 over time
        if (age <= 3 hours) return 10_000;
        if (age <= 24 hours) return 6_500;
        if (age <= 3 days) return 3_500;
        return 1_500;
    }
